⚙️ Como integrar nos outros sistemas

    
Para adicionar o download automático em qualquer sistema do triploohesigmalbpl, adicione esta função JavaScript e chame downloadAutomatico(titulo, autor, conteudo) sempre que uma obra for gerada:

    
// ── FUNÇÃO DE DOWNLOAD AUTOMÁTICO ──
// Adicione este bloco em qualquer HTML do ecossistema

function downloadAutomatico(titulo, autor, conteudo, categoria, destino) {
  var data = new Date().toLocaleDateString('pt-BR');
  var hora = new Date().toLocaleTimeString('pt-BR');
  var num  = 'OBR-' + Date.now().toString().slice(-6);

  var txt = [
    '══════════════════════════════════════════════════',
    titulo.toUpperCase(),
    '══════════════════════════════════════════════════',
    'Protocolo:  ' + num,
    'Autoria:    ' + (autor || '—'),
    'Categoria:  ' + (categoria || '—'),
    'Destino:    ' + (destino || '—'),
    'Gerado em:  ' + data + ' às ' + hora,
    '──────────────────────────────────────────────────',
    '',
    conteudo || '(sem conteúdo)',
    '',
    '══════════════════════════════════════════════════',
    'triploohesigmalbpl · W.S.P. S/A-ME'
  ].join('\n');

  // ── DOWNLOAD IMEDIATO NO DISPOSITIVO ──
  var blob = new Blob([txt], { type: 'text/plain;charset=utf-8' });
  var url  = URL.createObjectURL(blob);
  var a    = document.createElement('a');
  a.href     = url;
  a.download = num + '_' + titulo.replace(/\s+/g, '_').slice(0, 40) + '.txt';
  document.body.appendChild(a);
  a.click();
  document.body.removeChild(a);
  URL.revokeObjectURL(url);

  // ── SALVA NO LOCALSTORAGE TAMBÉM (redundância) ──
  try {
    var chave = 'obras_geradas';
    var lista = JSON.parse(localStorage.getItem(chave) || '[]');
    lista.unshift({ num, titulo, autor, categoria, destino, conteudo, em: data });
    localStorage.setItem(chave, JSON.stringify(lista));
  } catch(e) {}

  return num; // retorna o protocolo
}
